        import json
import os
import pyfiglet
from datetime import datetime

dollar_unit = "💲"
acct_bal = 0.0

products = []
prices = []
quantities = []
sales = []


data_file = "shopstock.json"

def save_data():
    data = {
        "products": products,
        "prices": prices,
        "quantities": quantities,
        "acct_bal": acct_bal,
        "sales": sales
    }

    with open(data_file, "w") as file:
        json.dump(data, file, indent=5)

def load_data():
    global acct_bal

    if os.path.exists(data_file):
        with open(data_file, "r") as file:
            data = json.load(file)

        products.extend(data["products"])
        prices.extend(data["prices"])
        quantities.extend(data["quantities"])
        acct_bal = data["acct_bal"]
        sales.extend(data.get("sales", []))

def add_product():
    print("\n")
    print("Fill in the product's details respectively below:")
    print("Type in 'done' to go to Main Menu when you have finished entering the new products details")
    
    running = True
    while running:

        name = input("NAME: ")
        if name.lower() == "done":
            print("All new products saved successsfully")
            break

        if name.isdigit():
            print("Product name cannot contain only numbers!")
            continue

        duplicate = False

        for product in products:
            if product.lower() == name.lower():
                duplicate = True
                break

        if duplicate:
            print("This products already exists!")
            continue

        
        #PRICE
        while True:
            try:
                price = float(input("PRICE: "))
                
            

            except ValueError:
                print("Please enter a valid number!")
                continue 

            if price < 0:
                print("Price cannot have a negative value!")
                continue
            
            break

        
        #QUANTITY
        while True:
            try:
                quantity = int(input("QUANTITY: "))

            except ValueError:
                print("Please enter a valid number!")
                continue

            if quantity < 0:
                print("Quantity value cannot be negative!")
                continue
            
            break
        
        products.append(name)
        prices.append(price)
        quantities.append(quantity)

        save_data()
            
        print(products)
        print(f"New product, {name} has been verified and saved!")
        


def remove_product():
    print("\n")
    print("Type in the index number of the product you wish to remove")
    print("Note: Product index number start from 0!")
    print("Type in 'done' to return to Product Management.")
    print()
        
    running = True
    while running:
        print(products)
        print(prices)
        print("\n")

        choice = input("> ")

        if choice.lower() == "done":
            break
        
        try:
            remove_product_choice = int(choice)

            if 0 <= remove_product_choice < len(products):

                products.pop(remove_product_choice)
                prices.pop(remove_product_choice)
                quantities.pop(remove_product_choice)

                save_data()

                print("Product removed successfully")
            else:
                print("Invalid Product Index!")
        
        except ValueError:
            print("Please enter a valid index number or 'done'!")
            continue

def search_product():
    print("Product Search")
    print("Type 'e' to exit this section")
    search = input("> ")

    found = False

    for i in range(len(products)):
        if products[i].lower() == search.lower():
            found = True      
            print(f"{products[i]} - {dollar_unit}{prices[i]} - Stock: {quantities[i]}")
            

    if not found:
        print("Product not found!")
        
    


def record_sale():
    global acct_bal
    print("\n")
    print("Sale Records")
    
    product_name = input("PRODUCT NAME: ").strip()

    found = False

    for i in range(len(products)):
        if products[i].lower() == product_name.lower():
            found = True
            break

    if not found:
        print("Product not found!")
        return
    
    print(f"PRODUCT: {products[i]}")
    print(f"PRICE PER UNIT: {dollar_unit}{prices[i]}")
    print(f"AVAILABLE STOCK: {quantities[i]}")

    while True:
        try:
            quantity_sold = int(input("QUANTITY SOLD: "))

        except ValueError:
            print("Enter a valid whole number!")
            continue

        if quantity_sold <= 0:
            print("Quantity sold must be greater than zero!")
            continue

        if quantity_sold > quantities[i]:
            print("Not enough stock available!")
            continue

        break
    total_sale = prices[i] * quantity_sold
    quantities[i] -= quantity_sold

    acct_bal += total_sale

    sale_record = {
    "product": products[i],
    "quantity": quantity_sold,
    "unit_price": prices[i],
    "total": total_sale,
    "date_time": datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    }

    sales.append(sale_record)


    print("\nSale completed successfully!")
    print(f"Product: {products[i]}")
    print(f"Quantity sold: {quantity_sold}")
    print(f"Total sale: {dollar_unit}{total_sale}")
    print(f"Remaining stock: {quantities[i]}")
    print(f"Updated account balance: {dollar_unit}{acct_bal}")

    save_data()
    

def low_stock_alerts():
    print("\n LOW-STOCK ALERTS")

    try:
        stock_limit = int(input("Enter low-stock limit: "))

    except ValueError:
        print("Please enter a valid number!")
        return

    if stock_limit < 0:
        print("Stock limit cannot be negative!")
        return

    found = False

    for i in range(len(products)):
        if quantities[i] <= stock_limit:
            
            found = True

            print(
                f"{products[i]} - Stock: {quantities[i]}"
                " - LOW STOCK!"
            )

    if not found:
        print("No products are at or below that stock limit.")


def business_reports():
    print("\n BUSINESS REPORTS")

    total_products = len(products)
    total_units = sum(quantities)
    stock_value = 0

    for i in range(len(products)):
        stock_value += prices[i] * quantities[i]

    print(f"Number of products: {total_products}")
    print(f"Total units in stock: {total_units}")
    print(f"Current stock value at selling prices: {dollar_unit}{stock_value:.2f}")
    print(f"Account balance: {dollar_unit}{acct_bal:.2f}")

def view_sales_history():
    print("\nSALES HISTORY")

    if not sales:
        print("No sales have been recorded yet.")
        return

    for i in range(len(sales)):
        print(f"\nSale #{i + 1}")
        print(f"Date and time: {sales[i].get('date_time', 'Not recorded')}")
        print(f"Product: {sales[i]['product']}")
        print(f"Quantity sold: {sales[i]['quantity']}")
        print(f"Price per unit: {dollar_unit}{sales[i]['unit_price']:.2f}")
        print(f"Total sale: {dollar_unit}{sales[i]['total']:.2f}")


def restock():
